Cámara + Python · Arduino Uno · OLED SSD1306 · 5 LEDs

Detector de Dedos

La cámara del computador ve tu mano. Python detecta qué dedos están levantados y se lo avisa al Arduino por el cable USB. El Arduino enciende un LED por dedo y muestra el nombre en la pantalla OLED.

Maqueta

En la protoboard, los 5 agujeros de cada columna están unidos entre sí (arriba y abajo del canal por separado). Los rieles rojo y azul recorren todo el largo. La detección de la mano ocurre en el computador: el único cable hacia él es el USB.

Cable por cable

DesdeHastaPin Uno
LED pulgarPata larga del LEDD2
LED índicePata larga del LEDD3
LED medioPata larga del LEDD4
LED anularPata larga del LEDD5
LED meñiquePata larga del LEDD6
OLED SDADatos I2CA4
OLED SCLReloj I2CA5
OLED VCCRiel rojo (+)5V
OLED GND y resistenciasRiel azul (−)GND

Cada LED lleva tres cosas: la pata larga al pin digital, la pata corta a una resistencia de 220 Ω, y la otra punta de la resistencia al riel azul. Si un LED no enciende, lo más probable es que esté al revés.

Materiales

  • Arduino Uno con cable USB
  • Pantalla OLED SSD1306 128x64 I2C (la que ya compraste)
  • 5 LEDs de 5 mm, del color que tengas
  • 5 resistencias de 220 Ω (rojo, rojo, café)
  • Protoboard de 830 puntos
  • Unos 15 cables dupont macho-macho
  • Computador con cámara web y Python 3

Ya no hacen falta sensores flex ni guante. La OLED vuelve a los pines I2C normales del Uno, A4 y A5.

Puesta en marcha

  1. En el Arduino IDE instala la biblioteca U8g2 y sube detector_dedos.ino. Los LEDs parpadean uno por uno al arrancar.
  2. Cierra el Monitor Serie del Arduino IDE: el puerto solo lo puede usar un programa a la vez.
  3. En una terminal instala las librerías: pip install opencv-python mediapipe pyserial
  4. Ejecuta python detector_dedos.py. Si no encuentra el Arduino, indícale el puerto: --puerto COM3
  5. Muestra tu mano a la cámara y levanta un dedo. Con la tecla Q se cierra.

Mira las letras impresas en tu OLED antes de cablear: algunos módulos traen VCC y GND en el orden contrario. Si el texto sale de cabeza, cambia U8G2_R2 por U8G2_R0 en el código.

Código del Arduino

/*
  DETECTOR DE DEDOS - Arduino Uno + OLED SSD1306 128x64 + 5 LEDs
  ---------------------------------------------------------------
  La camara y Python (OpenCV) detectan que dedos estan levantados y se lo
  mandan al Arduino por el cable USB. El Arduino enciende un LED por dedo
  y muestra el nombre en la pantalla.

  Libreria necesaria: "U8g2" de oliver
  (Arduino IDE > Herramientas > Administrar bibliotecas > buscar U8g2)

  CONEXIONES
    LED pulgar   -> D2
    LED indice   -> D3
    LED medio    -> D4
    LED anular   -> D5
    LED menique  -> D6
    (pata larga del LED al pin; pata corta a una resistencia de 220 ohm
     que va a GND)

    OLED VCC -> 5V
    OLED GND -> GND
    OLED SDA -> A4
    OLED SCL -> A5

  MENSAJES QUE LLEGAN DESDE PYTHON (9600 baudios, terminados en salto de linea)
    "01000"  cinco caracteres, uno por dedo (pulgar..menique): 1 = levantado
    "N"      no se ve ninguna mano
*/

#include <Arduino.h>
#include <Wire.h>
#include <U8g2lib.h>

// U8G2_R2 = imagen girada 180 grados (la OLED va con los pines hacia abajo
// en la protoboard). Si el texto te sale de cabeza, cambia U8G2_R2 por U8G2_R0.
U8G2_SSD1306_128X64_NONAME_1_HW_I2C oled(U8G2_R2, U8X8_PIN_NONE);

const uint8_t NUM_DEDOS = 5;
const uint8_t PIN_LED[NUM_DEDOS] = {2, 3, 4, 5, 6};
const char *const NOMBRE[NUM_DEDOS] = {"PULGAR", "ÍNDICE", "MEDIO", "ANULAR", "MEÑIQUE"};

// Si Python deja de hablar por este tiempo, se apagan los LEDs
const unsigned long TIEMPO_SIN_DATOS = 3000;

bool levantado[NUM_DEDOS];
bool hayMano = false;
bool conectado = false;
unsigned long ultimoMensaje = 0;

char linea[8];
uint8_t largo = 0;

// Titulo en la franja amarilla, texto grande en la azul
// y abajo una barra por dedo (alta = levantado).
void dibujar(const char *titulo, const char *texto) {
  oled.firstPage();
  do {
    oled.setFont(u8g2_font_6x12_tf);
    oled.drawUTF8((128 - oled.getUTF8Width(titulo)) / 2, 12, titulo);

    oled.setFont(u8g2_font_helvB14_tf);
    oled.drawUTF8((128 - oled.getUTF8Width(texto)) / 2, 38, texto);

    for (uint8_t i = 0; i < NUM_DEDOS; i++) {
      uint8_t alto = levantado[i] ? 18 : 4;
      oled.drawBox(8 + i * 24, 64 - alto, 16, alto);
    }
  } while (oled.nextPage());
}

void mostrarEstado() {
  uint8_t cuantos = 0;
  uint8_t ultimo = 0;
  for (uint8_t i = 0; i < NUM_DEDOS; i++) {
    digitalWrite(PIN_LED[i], levantado[i] ? HIGH : LOW);
    if (levantado[i]) {
      cuantos++;
      ultimo = i;
    }
  }

  char texto[12];
  if (!conectado) {
    dibujar("ESPERANDO A PYTHON", "SIN DATOS");
  } else if (!hayMano) {
    dibujar("MUESTRA TU MANO", "SIN MANO");
  } else if (cuantos == 0) {
    dibujar("NINGUN DEDO", "PUÑO");
  } else if (cuantos == 1) {
    dibujar("DEDO LEVANTADO", NOMBRE[ultimo]);
  } else if (cuantos == NUM_DEDOS) {
    dibujar("DEDOS LEVANTADOS", "TODOS");
  } else {
    snprintf(texto, sizeof(texto), "%u DEDOS", cuantos);
    dibujar("DEDOS LEVANTADOS", texto);
  }
}

void apagarTodo() {
  for (uint8_t i = 0; i < NUM_DEDOS; i++) {
    levantado[i] = false;
  }
}

// Interpreta una linea completa. Devuelve true si cambio algo.
bool procesarLinea() {
  bool nuevo[NUM_DEDOS] = {false, false, false, false, false};
  bool nuevaMano;

  if (largo == 1 && linea[0] == 'N') {
    nuevaMano = false;
  } else if (largo == NUM_DEDOS) {
    for (uint8_t i = 0; i < NUM_DEDOS; i++) {
      if (linea[i] != '0' && linea[i] != '1') {
        return false;   // linea con basura: se ignora
      }
      nuevo[i] = (linea[i] == '1');
    }
    nuevaMano = true;
  } else {
    return false;
  }

  ultimoMensaje = millis();
  bool cambio = !conectado || (nuevaMano != hayMano);
  for (uint8_t i = 0; i < NUM_DEDOS; i++) {
    if (nuevo[i] != levantado[i]) {
      cambio = true;
    }
    levantado[i] = nuevo[i];
  }
  hayMano = nuevaMano;
  conectado = true;
  return cambio;
}

void setup() {
  for (uint8_t i = 0; i < NUM_DEDOS; i++) {
    pinMode(PIN_LED[i], OUTPUT);
  }
  Serial.begin(9600);
  oled.begin();

  // Prueba de LEDs: se encienden uno por uno al arrancar
  for (uint8_t i = 0; i < NUM_DEDOS; i++) {
    digitalWrite(PIN_LED[i], HIGH);
    delay(120);
    digitalWrite(PIN_LED[i], LOW);
  }
  mostrarEstado();
}

void loop() {
  bool cambio = false;

  while (Serial.available() > 0) {
    char c = Serial.read();
    if (c == '\n') {
      if (procesarLinea()) {
        cambio = true;
      }
      largo = 0;
    } else if (c != '\r') {
      if (largo < sizeof(linea) - 1) {
        linea[largo] = c;
      }
      if (largo < 255) {
        largo++;   // una linea demasiado larga queda con largo invalido y se ignora
      }
    }
  }

  if (conectado && millis() - ultimoMensaje > TIEMPO_SIN_DATOS) {
    conectado = false;
    hayMano = false;
    apagarTodo();
    cambio = true;
  }

  // Solo redibuja cuando algo cambia
  if (cambio) {
    mostrarEstado();
  }
}

Código de Python

"""
DETECTOR DE DEDOS - parte de Python

Lee la camara con OpenCV, detecta la mano con MediaPipe y le manda al Arduino
que dedos estan levantados. El Arduino enciende un LED por dedo y muestra el
nombre en la pantalla OLED.

La primera vez descarga el modelo de la mano (hand_landmarker.task, 8 MB)
y lo guarda junto a este archivo, asi que necesita internet esa vez.

Instalar:   pip install -r requirements.txt
Ejecutar:   python detector_dedos.py
            python detector_dedos.py --puerto COM3      (Windows)
            python detector_dedos.py --puerto /dev/ttyACM0   (Linux)
Salir:      tecla Q sobre la ventana de video

IMPORTANTE: cierra el Monitor Serie del Arduino IDE antes de ejecutar,
porque el puerto solo puede usarlo un programa a la vez.
"""

import argparse
import math
import time
import urllib.request
from pathlib import Path

import cv2
import mediapipe as mp
import serial
from serial.tools import list_ports

NOMBRES = ["PULGAR", "INDICE", "MEDIO", "ANULAR", "MENIQUE"]
BAUDIOS = 9600
CUADROS_ESTABLES = 3      # el estado debe repetirse estos cuadros antes de enviarse
SEGUNDOS_REENVIO = 1.0    # se reenvia cada segundo para que el Arduino sepa que seguimos aqui

MODELO = Path(__file__).with_name("hand_landmarker.task")
URL_MODELO = (
    "https://storage.googleapis.com/mediapipe-models/hand_landmarker/"
    "hand_landmarker/float16/1/hand_landmarker.task"
)

# Lineas que unen los 21 puntos de la mano (para dibujarla sobre el video)
HUESOS = [
    (0, 1), (1, 2), (2, 3), (3, 4),
    (0, 5), (5, 6), (6, 7), (7, 8),
    (5, 9), (9, 10), (10, 11), (11, 12),
    (9, 13), (13, 14), (14, 15), (15, 16),
    (13, 17), (17, 18), (18, 19), (19, 20), (0, 17),
]

# Puntos de la mano segun MediaPipe
MUNECA = 0
PULGAR_NUDILLO, PULGAR_PUNTA = 3, 4
BASE_MENIQUE = 17
# (punta, articulacion media) de indice, medio, anular y menique
DEDOS_LARGOS = [(8, 6), (12, 10), (16, 14), (20, 18)]


def distancia(a, b):
    return math.hypot(a.x - b.x, a.y - b.y)


def dedos_levantados(puntos):
    """Devuelve 5 valores True/False: pulgar, indice, medio, anular, menique.

    Compara distancias en vez de alturas, asi funciona aunque la mano
    este inclinada y con cualquiera de las dos manos.
    """
    muneca = puntos[MUNECA]
    base_menique = puntos[BASE_MENIQUE]

    # Pulgar: estirado si la punta se aleja del borde opuesto de la palma
    estado = [
        distancia(puntos[PULGAR_PUNTA], base_menique)
        > distancia(puntos[PULGAR_NUDILLO], base_menique)
    ]
    # Resto: estirado si la punta queda mas lejos de la muneca que su articulacion
    for punta, media in DEDOS_LARGOS:
        estado.append(distancia(puntos[punta], muneca) > distancia(puntos[media], muneca))
    return estado


def a_mensaje(estado):
    """None = sin mano -> 'N'.  Lista de 5 -> por ejemplo '01000'."""
    if estado is None:
        return "N"
    return "".join("1" if e else "0" for e in estado)


def crear_detector():
    if not MODELO.exists():
        print("Descargando el modelo de la mano (solo la primera vez)...")
        urllib.request.urlretrieve(URL_MODELO, MODELO)
    vision = mp.tasks.vision
    opciones = vision.HandLandmarkerOptions(
        base_options=mp.tasks.BaseOptions(model_asset_path=str(MODELO)),
        running_mode=vision.RunningMode.VIDEO,
        num_hands=1,
        min_hand_detection_confidence=0.6,
        min_tracking_confidence=0.6,
    )
    return vision.HandLandmarker.create_from_options(opciones)


def dibujar_mano(cuadro, puntos):
    alto, ancho = cuadro.shape[:2]
    pix = [(int(p.x * ancho), int(p.y * alto)) for p in puntos]
    for a, b in HUESOS:
        cv2.line(cuadro, pix[a], pix[b], (255, 255, 255), 2)
    for punto in pix:
        cv2.circle(cuadro, punto, 4, (0, 200, 255), -1)


def buscar_puerto():
    """Busca un puerto que parezca un Arduino."""
    pistas = ("arduino", "ch340", "usb serial", "usb-serial", "ttyacm", "ttyusb", "usbmodem")
    for p in list_ports.comports():
        texto = f"{p.device} {p.description} {p.manufacturer or ''}".lower()
        if any(pista in texto for pista in pistas):
            return p.device
    return None


def abrir_arduino(puerto):
    puerto = puerto or buscar_puerto()
    if puerto is None:
        print("No encontre el Arduino. Sigo solo con la camara.")
        print("Puertos disponibles:", [p.device for p in list_ports.comports()] or "ninguno")
        print("Puedes indicarlo con:  --puerto COM3")
        return None
    try:
        arduino = serial.Serial(puerto, BAUDIOS, timeout=0, write_timeout=1)
    except serial.SerialException as error:
        print(f"No pude abrir {puerto}: {error}")
        print("Revisa que el Monitor Serie del Arduino IDE este cerrado. Sigo solo con la camara.")
        return None
    print(f"Arduino conectado en {puerto}. Esperando a que reinicie...")
    time.sleep(2)   # el Uno se reinicia cuando se abre el puerto
    return arduino


def main():
    parser = argparse.ArgumentParser(description="Detecta dedos levantados y avisa al Arduino")
    parser.add_argument("--puerto", help="puerto serie del Arduino (ej: COM3 o /dev/ttyACM0)")
    parser.add_argument("--camara", type=int, default=0, help="numero de camara (0 = la principal)")
    args = parser.parse_args()

    arduino = abrir_arduino(args.puerto)

    camara = cv2.VideoCapture(args.camara)
    if not camara.isOpened():
        raise SystemExit(f"No pude abrir la camara {args.camara}. Prueba con --camara 1")

    detector = crear_detector()
    inicio = time.monotonic()

    candidato, repeticiones = None, 0
    enviado, ultimo_envio = None, 0.0

    try:
        while True:
            ok, cuadro = camara.read()
            if not ok:
                print("La camara dejo de entregar imagen.")
                break

            cuadro = cv2.flip(cuadro, 1)   # efecto espejo
            imagen = mp.Image(
                image_format=mp.ImageFormat.SRGB, data=cv2.cvtColor(cuadro, cv2.COLOR_BGR2RGB)
            )
            milisegundos = int((time.monotonic() - inicio) * 1000)
            resultado = detector.detect_for_video(imagen, milisegundos)

            estado = None
            if resultado.hand_landmarks:
                puntos = resultado.hand_landmarks[0]
                dibujar_mano(cuadro, puntos)
                estado = dedos_levantados(puntos)

            # Filtro: el mensaje debe repetirse unos cuadros para evitar parpadeos
            mensaje = a_mensaje(estado)
            if mensaje == candidato:
                repeticiones += 1
            else:
                candidato, repeticiones = mensaje, 1

            ahora = time.time()
            estable = repeticiones >= CUADROS_ESTABLES
            if estable and (candidato != enviado or ahora - ultimo_envio > SEGUNDOS_REENVIO):
                if candidato != enviado:
                    print("Enviado:", candidato)
                enviado, ultimo_envio = candidato, ahora
                if arduino is not None:
                    try:
                        arduino.write((candidato + "\n").encode("ascii"))
                    except serial.SerialException as error:
                        print(f"Se perdio la conexion con el Arduino: {error}")
                        arduino = None

            # Texto sobre el video
            if estado is None:
                texto = "Sin mano"
            else:
                arriba = [NOMBRES[i] for i, e in enumerate(estado) if e]
                texto = ", ".join(arriba) if arriba else "Puno"
            cv2.putText(cuadro, texto, (12, 36), cv2.FONT_HERSHEY_SIMPLEX, 0.9, (0, 0, 0), 5)
            cv2.putText(cuadro, texto, (12, 36), cv2.FONT_HERSHEY_SIMPLEX, 0.9, (0, 255, 255), 2)

            cv2.imshow("Detector de dedos (Q para salir)", cuadro)
            if cv2.waitKey(1) & 0xFF in (ord("q"), ord("Q")):
                break
    finally:
        camara.release()
        cv2.destroyAllWindows()
        detector.close()
        if arduino is not None:
            try:
                arduino.write(b"N\n")
                arduino.close()
            except serial.SerialException:
                pass


if __name__ == "__main__":
    main()
